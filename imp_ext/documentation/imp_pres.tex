	\section{Syntax \& Semantics}
	\begin{frame}[fragile]{Abstract Syntax}
\begin{lstlisting}[language=Haskell]
data Stmt
  = Assign String Exp
  | Seq Stmt Stmt
  | If Exp Stmt Stmt
  | While Exp Stmt
  | Block Stmt        -- New scope
  | Par Stmt Stmt     -- Parallelism
  deriving (Show, Eq)
\end{lstlisting}
	\end{frame}

	\begin{frame}{Small-Step Semantics}
		We implemented small-step semantics to handle concurrency:
		\[
			\frac{\langle s_1, \sigma \rangle \rightarrow \langle s_1', \sigma' \rangle}
			{\langle s_1 \parallel s_2, \sigma \rangle \rightarrow \langle s_1' \parallel s_2, \sigma' \rangle}
		\]
		This allows generic interleaving of instructions.
	\end{frame}
